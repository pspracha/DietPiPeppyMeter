This is for 1280x400 configuration


***** Setup display with resolution of 1280x400 in dietpi

Step 1: Replace Chromium command with xinit ~/.xinitrc
sudo nano /var/lib/dietpi/dietpi-software/installed/chromium-autostart.sh

Find this line near the bottom:
exec "$STARTX" "$FP_CHROMIUM" $CHROMIUM_OPTS "${URL:-https://dietpi.com/}"

Replace it with:
exec "$STARTX" /home/dietpi/.xinitrc

Step 2: Create /home/dietpi/.xinitrc with your full setup
nano /home/dietpi/.xinitrc

Paste:
#!/bin/bash
export DISPLAY=:0
sleep 2

# Set custom resolution
xrandr --newmode "1280x400_60.00" 40.00 1280 1320 1440 1600 400 403 413 417 -hsync +vsync
xrandr --addmode HDMI-1 1280x400_60.00
xrandr -s 1280x400

# Chromium settings from dietpi.txt
RES_X=$(sed -n '/^[[:blank:]]*SOFTWARE_CHROMIUM_RES_X=/{s/^[^=]*=//p;q}' /boot/dietpi.txt)
RES_Y=$(sed -n '/^[[:blank:]]*SOFTWARE_CHROMIUM_RES_Y=/{s/^[^=]*=//p;q}' /boot/dietpi.txt)
URL=$(sed -n '/^[[:blank:]]*SOFTWARE_CHROMIUM_AUTOSTART_URL=/{s/^[^=]*=//p;q}' /boot/dietpi.txt)

CHROMIUM_OPTS="--kiosk --window-size=${RES_X:-1280},${RES_Y:-720} --window-position=0,0"
FP_CHROMIUM=$(command -v chromium-browser)
[ "$FP_CHROMIUM" ] || FP_CHROMIUM=$(command -v chromium)

exec "$FP_CHROMIUM" $CHROMIUM_OPTS "${URL:-https://dietpi.com/}"
Make it executable:

chmod +x /home/dietpi/.xinitrc

Reboot
